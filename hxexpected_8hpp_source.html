24template<typename T_>
25concept hxexpected_concept_ = requires(T_& x_, const T_& y_) {
26 sizeof(T_);
27 T_();
28 T_(y_);
29 T_(hxmove(x_));
30 x_ = y_;
31 x_ = hxmove(x_);
32 x_.~T_();
33 { x_ == y_ } -> hxsame_as<bool>;
34};
35
36template<typename E_>
37concept hxunexpected_concept_ = requires(E_& x_, const E_& y_) {
38 sizeof(E_);
39 E_(true);
40 E_(false);
41 E_(y_);
42 E_(hxmove(x_));
43 x_ = y_;
44 x_ = hxmove(x_);
45 x_.~E_();
46 { static_cast<bool>(y_) };
47 { x_ == y_ } -> hxsame_as<bool>;
48};
