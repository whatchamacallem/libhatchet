134// xxhash32 prime constants and avalanche mixing.
135hxinline_constexpr hxhash_t hxhash_k1_ = hxhash_t{0x9E3779B1u};
136hxinline_constexpr hxhash_t hxhash_k2_ = hxhash_t{0x85EBCA77u};
137hxinline_constexpr hxhash_t hxhash_k3_ = hxhash_t{0xC2B2AE3Du};
138hxinline_constexpr hxhash_t hxhash_k4_ = hxhash_t{0x27D4EB2Fu};
139hxinline_constexpr hxhash_t hxhash_k5_ = hxhash_t{0x165667B1u};
140
141// xxhash32 avalanche.
142hxattr_nodiscard hxinline hxconstexpr
143hxhash_t hxhash_avalanche_(hxhash_t x_) {
144 x_ ^= x_ >> 15u;
145 x_ *= hxhash_k2_;
146 x_ ^= x_ >> 13u;
147 x_ *= hxhash_k3_;
148 x_ ^= x_ >> 16u;
149 return x_;
150}

158template<typename T_, typename enabled_t=void>
159class hxkey_hash_t {
160public:
161 hxattr_nodiscard hxinline hxhash_t operator()(const T_& x_) const {
162 // xxhash32 short-input path: seed=0, length=4, single 4-byte word.
163 hxhash_t h_ = hxhash_k5_ + hxhash_t{4u};
164 // Did you write a custom hxkey_hash_t specialization?
165 h_ += static_cast<hxhash_t>(x_) * hxhash_k3_;
166 h_ = ((h_ << 17u) | (h_ >> 15u)) * hxhash_k4_;
167 return hxhash_avalanche_(h_);
168 }
169};

174template<typename T_>
175class hxkey_hash_t<T_, hxenable_if_t<hxis_string<T_>()>> {
176public:
177 hxattr_nodiscard hxinline hxhash_t operator()(const volatile char* s_) const {
178 const char* str_ = const_cast<const char*>(s_);
179 hxhash_t h_ = hxhash_k5_;
180 while(*str_ != '\0') {
181 h_ += static_cast<hxhash_t>(static_cast<unsigned char>(*str_++)) * hxhash_k5_;
182 h_ = ((h_ << 11u) | (h_ >> 21u)) * hxhash_k1_;
183 }
184 return hxhash_avalanche_(h_);
185 }
