120// xxhash32 prime constants and avalanche mixing. Useful when hashing sequential
121// data. These are used by hxkey_hash_t specializations below.
122hxinline_constexpr hxhash_t hxhash_prime1_ = hxhash_t{0x9E3779B1u};
123hxinline_constexpr hxhash_t hxhash_prime2_ = hxhash_t{0x85EBCA77u};
124hxinline_constexpr hxhash_t hxhash_prime3_ = hxhash_t{0xC2B2AE3Du};
125hxinline_constexpr hxhash_t hxhash_prime4_ = hxhash_t{0x27D4EB2Fu};
126hxinline_constexpr hxhash_t hxhash_prime5_ = hxhash_t{0x165667B1u};
127
128// xxhash32 avalanche: x ^= x >> 15, x *= prime2, x ^= x >> 13, x *= prime3, x ^= x >> 16.
129hxattr_nodiscard hxinline hxconstexpr
130hxhash_t hxhash_avalanche_(hxhash_t x_) {
131 x_ ^= x_ >> 15u;
132 x_ *= hxhash_prime2_;
133 x_ ^= x_ >> 13u;
134 x_ *= hxhash_prime3_;
135 x_ ^= x_ >> 16u;
136 return x_;
137}

145template<typename T_, typename enabled_t=void>
146class hxkey_hash_t {
147public:
148 hxattr_nodiscard hxinline hxhash_t operator()(const T_& x_) const {
149 // xxhash32 short-input path: seed=0, length=4, single 4-byte word.
150 hxhash_t h_ = hxhash_prime5_ + hxhash_t{4u};
151 // Did you write a custom hxkey_hash_t specialization?
152 h_ += static_cast<hxhash_t>(x_) * hxhash_prime3_;
153 h_ = ((h_ << 17u) | (h_ >> 15u)) * hxhash_prime4_;
154 return hxhash_avalanche_(h_);
155 }
156};

161template<typename T_>
162class hxkey_hash_t<T_, hxenable_if_t<hxis_string<T_>()>> {
163public:
164 hxattr_nodiscard hxinline hxhash_t operator()(const volatile char* s_) const {
165 const char* t_ = const_cast<const char*>(s_);
166 hxhash_t h_ = hxhash_prime5_;
167 while(*t_ != '\0') {
168 h_ += static_cast<hxhash_t>(static_cast<unsigned char>(*t_++)) * hxhash_prime5_;
169 h_ = ((h_ << 11u) | (h_ >> 21u)) * hxhash_prime1_;
170 }
171 return hxhash_avalanche_(h_);
172 }
