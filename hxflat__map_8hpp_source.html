538 class sort_value_ {
539 public:
540 // Refers to the pointed-to key and mapped value in place. m_key_val_
541 // and m_mapped_val_ remain raw memory.
542 sort_value_(key_t_* key_, mapped_t_* mapped_);
543 sort_value_(const sort_value_&) = default;
544
545 // Detaches into owned storage and takes over as the live instance.
546 sort_value_(sort_value_&& sv_) noexcept;
547
548 ~sort_value_(void);
549
550 void operator=(sort_value_&& sv_) noexcept;
551 bool operator<(const sort_value_& x_) const;
552
553 friend void hxswap(const sort_value_& a_, const sort_value_& b_) {
554 hxswap(*a_.m_key_, *b_.m_key_);
555 hxswap(*a_.m_mapped_, *b_.m_mapped_);
556 }
557
558 private:
559 // Raw storage unless m_key_ and m_mapped_ point at it.
560 union { key_t_ m_key_val_; };
561 union { mapped_t_ m_mapped_val_; };
