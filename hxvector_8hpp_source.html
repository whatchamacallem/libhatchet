38// Internal. Extends the array using placement new when assigned to.
39template<typename array_t_>
40class hxarray_back_inserter_ {
41public:
42 // Internal. Extends the array using placement new when assigned to.
43 template<typename arg_t_>
44 typename array_t_::value_t& operator=(arg_t_&& arg_) {
45 return m_x_.push_back(hxforward<arg_t_>(arg_));
46 }
47private:
48 friend array_t_;
49 hxarray_back_inserter_(array_t_& x_) : m_x_(x_) { }
50#if HX_CPLUSPLUS >= 201703L
51 // Copy elision is C++17.
52 hxarray_back_inserter_(const hxarray_back_inserter_& x_) = delete;
53#endif
54 // No address-of operator. It wouldn't be what was expected.
55 void operator&(void) const = delete;
56 array_t_& m_x_;
57};
