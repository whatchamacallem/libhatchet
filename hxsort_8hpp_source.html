33template<hxrandom_iterator_concept_ iterator_t_, typename less_t_>
34hxinline hxconstexpr hxattr_flatten
35void hxinsertion_sort(iterator_t_ begin_, iterator_t_ end_, const less_t_& less_) {
36 // Address sanitizer: Avoids adding 1 to null iterators.
37 if(begin_ == end_) { return; }
38 const hxrestrict_t<iterator_t_> begin_r_(begin_);
39 for(iterator_t_ i_ = begin_r_, j_ = begin_r_ + ptrdiff_t{1}; j_ < end_; i_ = j_, ++j_) {
40 if(less_(*j_, *i_)) {
41 auto t_ = hxmove(*j_);
42 *j_ = hxmove(*i_);
43 while(begin_r_ < i_ && less_(t_, *(i_ - ptrdiff_t{1}))) {
44 *i_ = hxmove(*(i_ - ptrdiff_t{1}));
45 --i_;
46 }
47 *i_ = hxmove(t_);
48 }
49 }
50}

64template<hxrandom_iterator_concept_ iterator_t_, typename less_t_>
65hxinline hxconstexpr void hxheapsort(iterator_t_ begin_, iterator_t_ end_, const less_t_& less_) {
66 if(begin_ == end_) { return; } // Prevents UB.
67 hxdetail_::hxmake_heap_<iterator_t_>(begin_, end_, less_);
68 const hxrestrict_t<iterator_t_> begin_r_(begin_);
69 for(iterator_t_ it_ = end_ - ptrdiff_t{1}; begin_r_ < it_; --it_) {
70 hxswap(*begin_r_, *it_);
71 hxdetail_::hxheapsort_heapify_<iterator_t_>(begin_r_, begin_r_, it_, less_);
72 }
73}
