104 hxattr_nodiscard hxinline hxconstexpr
105 uint32_t u32(void) {
106 m_state_ = uint64_t{0x5851f42d4c957f2dull} * m_state_ + uint64_t{0x14057b7ef767814full};
107
108 // MODIFICATION: Use the 4 msb bits as a random 0..15 bit variable shift
109 // control. Ignores the low 13 bits because they are low quality.
110 // Returns 32 bits chosen at a random offset starting between the 13th
111 // and 28th bits. 4 bits shift control + 32 returned + up to 15 shifted
112 // off + 13 always discarded = 64 bits.
113 const uint32_t result_ = static_cast<uint32_t>(m_state_ >> ((m_state_ >> 60) + 13u));
114 return result_;
115 }
