36template<typename T_>
37class hxthread_local {
38 static_assert(sizeof(T_) <= sizeof(void*),
39 "hxthread_local: sizeof(T) must be <= sizeof(void*)");
40
41public:
43 hxinline hxattr_flatten explicit hxthread_local(void) {
44#if (HX_USE_THREADS) == 11
45 const int code_ = ::tss_create(&m_key_, 0);
46 hxassert_always(code_ == thrd_success, "tss_create %d", code_); (void)code_;
47#elif HX_USE_THREADS
48 const int code_ = ::pthread_key_create(&m_key_, 0);
49 hxassert_always(code_ == 0, "pthread_key_create %s", ::strerror(code_)); (void)code_;
50#else
51 m_value_ = T_();
52#endif
53 }
54
56 hxinline hxattr_flatten ~hxthread_local() {
57#if (HX_USE_THREADS) == 11
58 ::tss_delete(m_key_);
59#elif HX_USE_THREADS
60 ::pthread_key_delete(m_key_);
61#endif
62 }
63
65 hxinline hxattr_flatten operator T_(void) {
66#if (HX_USE_THREADS) == 11
67 return (T_)(intptr_t)::tss_get(m_key_); // NOLINT(google-readability-casting)
68#elif HX_USE_THREADS
69 return (T_)(intptr_t)::pthread_getspecific(m_key_); // NOLINT(google-readability-casting)
70#else
71 return m_value_;
72#endif
73 }
74
76 hxinline hxattr_flatten void operator=(T_ local_) {
77#if (HX_USE_THREADS) == 11
78 const int code_ = ::tss_set(m_key_, (void*)(intptr_t)local_); // NOLINT(google-readability-casting)
79 hxassert_hard(code_ == thrd_success, "tss_set %d", code_); (void)code_;
80#elif HX_USE_THREADS
81 const int code_ = ::pthread_setspecific(m_key_, (void*)(intptr_t)local_); // NOLINT(google-readability-casting)
82 hxassert_hard(code_ == 0, "pthread_setspecific %s", ::strerror(code_)); (void)code_;
83#else
84 m_value_ = local_;
85#endif
86 }
87
88private:
89 explicit hxthread_local(const hxthread_local&) = delete;
90 hxthread_local& operator=(const hxthread_local&) = delete;
91
92 // No allocation is being made with which to return a pointer to.
93 void operator&(void) const = delete;
94 void operator&(void) = delete;
95
96#if (HX_USE_THREADS) == 11
97 ::tss_t m_key_;
98#elif HX_USE_THREADS
99 ::pthread_key_t m_key_;
100#else
101 T_ m_value_;
102#endif
103};

202public:
205 hxinline hxunique_lock(hxmutex& mtx_, bool defer_lock_=false)
206 : m_mutex_(mtx_), m_owns_(false) {
207 if(!defer_lock_) {
208 this->lock();
209 }
210 }
211
212 hxinline ~hxunique_lock(void) {
213 if(m_owns_) {
214 this->unlock();
215 }
216 }
217
218 hxinline void lock(void) {
219 if(!m_owns_) {
220 m_owns_ = m_mutex_.lock();
221 }
222 }
223
224 hxinline void unlock(void) {
225 if(m_owns_) {
226 m_mutex_.unlock();
227 m_owns_ = false;
228 }
229 }
230
232 hxinline bool owns_lock(void) const { return m_owns_; }
233
235 hxinline hxmutex& mutex(void) { return m_mutex_; }

237private:

369 template<typename parameter_t_>
370 hxinline explicit hxthread(return_t (*entry_point_)(parameter_t_*), parameter_t_* parameter_)
371 : hxthread() {
372 this->start(entry_point_, parameter_);
373 }
374
376 hxinline ~hxthread(void) {
377 hxassert_hard(!this->joinable(), "thread_busy");
378 }
379
386 template<typename parameter_t_>
387 hxinline void start(return_t (*entry_point_)(parameter_t_*), parameter_t_* parameter_) {
388 hxassertf(!this->joinable(), "thread_busy");
389
390 // Stay on the right side of the C++ standard by avoiding assumptions
391 // about pointer representations. The parameter is being reinterpreted
392 // twice instead of cast once and reinterpreted back.
393 static_assert(sizeof(void*) == sizeof(parameter_t_*), "Incompatible pointer sizes");
394
395 void* reinterpreted_parameter_ = hxnull;
396 ::memcpy(&reinterpreted_parameter_, &parameter_, sizeof(void*)); // NOLINT(bugprone-bitwise-pointer-cast)
397#if (HX_USE_THREADS) == 11
398 const int code_ = ::thrd_create(&m_thread_,
399 reinterpret_cast<entry_point_function_t_>(entry_point_), reinterpreted_parameter_);
400 hxassert_always(code_ == thrd_success, "thrd_create %d", code_); (void)code_;
401#else
402 const int code_ = ::pthread_create(&m_thread_, 0,
403 reinterpret_cast<entry_point_function_t_>(entry_point_), reinterpreted_parameter_);
404 hxassert_always(code_ == 0, "pthread_create %s", ::strerror(code_)); (void)code_;
405#endif
406 m_joinable_ = true;
407 }
408
411 hxinline hxattr_nodiscard bool joinable(void) const { return m_joinable_; }
412
414 hxinline void join(void) {
415 hxassertf(this->joinable(), "task_idle");
416#if (HX_USE_THREADS) == 11
417 const int code_ = ::thrd_join(m_thread_, hxnull);
418 hxassert_always(code_ == thrd_success, "thrd_join %d", code_);
419 (void)code_;
420#else
421 const int code_ = ::pthread_join(m_thread_, 0);
422 hxassert_always(code_ == 0, "pthread_join %s", ::strerror(code_));
423 (void)code_;
424#endif
425 m_joinable_ = false;
426 }
427
428private:
429 using entry_point_function_t_ = return_t (*)(void*);
430
431 hxthread(const hxthread&) = delete;
432 hxthread& operator=(const hxthread&) = delete;
433
434#if (HX_USE_THREADS) == 11
435 ::thrd_t m_thread_;
436#else
437 ::pthread_t m_thread_;
438#endif
439 bool m_joinable_;
440};
