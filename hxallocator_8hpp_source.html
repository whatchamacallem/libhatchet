76template<typename T_, bool pow2_>
77class hxallocator<T_, hxallocator_dynamic_capacity, pow2_> {
78public:
80 using value_t = T_;
81
83 hxinline hxallocator(void) {
84 m_data_ = hxnull;
85 m_capacity_ = 0;
86 }
87
89 hxinline ~hxallocator(void) {
90 if(m_data_) {
91 m_capacity_ = 0;
92 hxfree(m_data_);
93 m_data_ = hxnull;
94 }
95 }
96
99 hxinline hxattr_nodiscard hxsize_t capacity(void) const { return m_capacity_; }
100
102 hxinline const T_* data(void) const { return m_data_; }
103
105 hxinline T_* data(void) { return m_data_; }
106
115 hxinline void reserve_storage(hxsize_t size_,
116 hxslab_allocator_t allocator_=hxslab_allocator_current,
117 hxalignment_t alignment_=hxalignment) {
118 if(size_ == m_capacity_) {
119 return;
120 }
121 hxassert_always(m_capacity_ == 0, "bad_reserve already sized");
122 const hxsize_t element_count_ = pow2_ ? (static_cast<hxsize_t>(1) << size_) : size_;
123 m_data_ = static_cast<T_*>(
124 hxmalloc_ext(sizeof(T_) * static_cast<size_t>(element_count_), allocator_, alignment_));
125 m_capacity_ = size_;
126 }
127
128private:
129 hxallocator(const hxallocator&) = delete;
130 void operator=(const hxallocator&) = delete;
131
132 hxsize_t m_capacity_;
133 T_* m_data_;
134};
