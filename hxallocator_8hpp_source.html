109 hxsystem_allocator_t allocator_=hxsystem_allocator_current,
110 hxalignment_t alignment_=hxalignment) {
111 if(size_ == m_capacity_) {
112 return;
113 }
114 hxassert_always(m_capacity_ == 0, "bad_reserve already sized");
115 m_data_ = static_cast<T_*>(
