template<typename key_t_, typename mapped_t_>
class hxflat_map_const_value_t< key_t_, mapped_t_ >

hxflat_map_const_value_t - A proxy referencing the key and mapped value at a fixed position, returned by dereferencing an hxflat_map's const_iterator.
