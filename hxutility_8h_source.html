312hxattr_nodiscard hxinline int hxlog2i(uint32_t i_) {
313#if defined _MSC_VER
314 unsigned long index_ = 0ul;
315 ::_BitScanReverse(&index_, i_);
316 return static_cast<int>(index_);
317#elif defined __GNUC__ || defined __clang__
318 return 31 - __builtin_clz(i_);
319#else
320 float f_ = static_cast<float>(i_);
321 uint32_t bits_ = 0u; ::memcpy(&bits_, &f_, sizeof f_);
322 return static_cast<int>((bits_ >> 23) & 0xffu) - 127;
323#endif
324}
