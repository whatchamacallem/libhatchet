33template<hxrandom_range_concept_ range0_t_, hxrandom_range_concept_ range1_t_,
34 typename output_iterator_t_, typename less_t_>
35inline hxconstexpr hxattr_flatten
36output_iterator_t_ hxmerge(range0_t_&& range0_, range1_t_&& range1_,
37 output_iterator_t_&& output_, const less_t_& less_) noexcept {
38 hxrestrict_t<decltype(range0_.begin())> src0_(range0_.begin());
39 const auto end0_ = range0_.end();
40 hxrestrict_t<decltype(range1_.begin())> src1_(range1_.begin());
41 const auto end1_ = range1_.end();
42 hxrestrict_t<output_iterator_t_> output_r_(hxforward<output_iterator_t_>(output_));
43 hxassert_different_iterator(output_r_, src0_);
44 hxassert_different_iterator(output_r_, src1_);
45 while(src0_ != end0_ && src1_ != end1_) {
46 const bool take1_ = less_(*src1_, *src0_);
47 if(take1_) { *output_r_ = hxforward_like<range1_t_>(*src1_); }
48 else { *output_r_ = hxforward_like<range0_t_>(*src0_); }
49 ++output_r_;
50 src1_ = src1_ + static_cast<ptrdiff_t>(take1_);
51 src0_ = src0_ + static_cast<ptrdiff_t>(!take1_);
52 }
53 while(src0_ != end0_) {
54 *output_r_ = hxforward_like<range0_t_>(*src0_);
55 ++output_r_; ++src0_;
56 }
57 while(src1_ != end1_) {
58 *output_r_ = hxforward_like<range1_t_>(*src1_);
59 ++output_r_; ++src1_;
60 }
61 return output_r_;
62}

93template<hxrange_concept_ range_t_, typename less_t_>
94hxattr_nodiscard inline hxconstexpr hxattr_flatten
95auto hxminmax(range_t_&& range_, const less_t_& less_) -> hxminmax_result<decltype(range_.begin())> {
96 hxrestrict_t<decltype(range_.begin())> it_(range_.begin());
97 const auto end_ = range_.end();
98 hxminmax_result<decltype(range_.begin())> result_{it_, it_};
99 if(it_ == end_) { return result_; }
100 for(++it_; it_ != end_; ++it_) {
101 if(less_(*it_, *result_.min)) {
102 result_.min = it_;
103 }
104 else if(less_(*result_.max, *it_)) {
105 result_.max = it_;
106 }
107 }
108 return result_;
109}

285template<hxrandom_iterator_concept_ iterator_t_, typename equal_t_>
286hxattr_nodiscard inline hxconstexpr hxattr_flatten
287iterator_t_ hxunique(iterator_t_ begin_, iterator_t_ end_, const equal_t_& equal_) noexcept {
288 if(begin_ == end_) { return end_; }
289 iterator_t_ dst_ = begin_;
290 for(++begin_; begin_ != end_; ++begin_, ++dst_) {
291 if(equal_(*dst_, *begin_)) { break; }
292 }
293 if(begin_ == end_) { return end_; }
294 hxrestrict_t<iterator_t_> dst_r_(dst_);
295 hxrestrict_t<iterator_t_> src_r_(begin_);
296 for(++src_r_; src_r_ != end_; ++src_r_) {
297 if(!equal_(*dst_r_, *src_r_)) {
298 ++dst_r_;
299 *dst_r_ = hxmove(*src_r_);
300 }
301 }
302 return ++dst_r_;
303}
