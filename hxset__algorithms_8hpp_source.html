33template<hxrandom_range_concept_ range0_t_, hxrandom_range_concept_ range1_t_,
34 typename output_iterator_t_, typename less_t_>
35inline hxconstexpr hxattr_flatten
36output_iterator_t_ hxmerge(range0_t_&& range0_, range1_t_&& range1_,
37 output_iterator_t_&& output_, const less_t_& less_) noexcept {
38 hxrestrict_t<decltype(range0_.begin())> src0_(range0_.begin());
39 const auto end0_ = range0_.end();
40 hxrestrict_t<decltype(range1_.begin())> src1_(range1_.begin());
41 const auto end1_ = range1_.end();
42 hxrestrict_t<output_iterator_t_> output_r_(hxforward<output_iterator_t_>(output_));
43 while(src0_ != end0_ && src1_ != end1_) {
44 const bool take1_ = less_(*src1_, *src0_);
45 if(take1_) { *output_r_ = hxforward_like<range1_t_>(*src1_); }
46 else { *output_r_ = hxforward_like<range0_t_>(*src0_); }
47 ++output_r_;
48 src1_ = src1_ + static_cast<ptrdiff_t>(take1_);
49 src0_ = src0_ + static_cast<ptrdiff_t>(!take1_);
50 }
51 while(src0_ != end0_) {
52 *output_r_ = hxforward_like<range0_t_>(*src0_);
53 ++output_r_; ++src0_;
54 }
55 while(src1_ != end1_) {
56 *output_r_ = hxforward_like<range1_t_>(*src1_);
57 ++output_r_; ++src1_;
58 }
59 return output_r_;
60}

91template<hxrange_concept_ range_t_, typename less_t_>
92hxattr_nodiscard inline hxconstexpr hxattr_flatten
93auto hxminmax(range_t_&& range_, const less_t_& less_) -> hxminmax_result<decltype(range_.begin())> {
94 hxrestrict_t<decltype(range_.begin())> it_(range_.begin());
95 const auto end_ = range_.end();
96 hxminmax_result<decltype(range_.begin())> result_{it_, it_};
97 if(it_ == end_) { return result_; }
98 for(++it_; it_ != end_; ++it_) {
99 if(less_(*it_, *result_.min)) {
100 result_.min = it_;
101 }
102 else if(less_(*result_.max, *it_)) {
103 result_.max = it_;
104 }
105 }
106 return result_;
107}

277template<hxrandom_iterator_concept_ iterator_t_, typename equal_t_>
278hxattr_nodiscard inline hxconstexpr hxattr_flatten
279iterator_t_ hxunique(iterator_t_ begin_, iterator_t_ end_, const equal_t_& equal_) noexcept {
280 if(begin_ == end_) { return end_; }
281 iterator_t_ dst_ = begin_;
282 for(++begin_; begin_ != end_; ++begin_, ++dst_) {
283 if(equal_(*dst_, *begin_)) { break; }
284 }
285 if(begin_ == end_) { return end_; }
286 hxrestrict_t<iterator_t_> dst_r_(dst_);
287 hxrestrict_t<iterator_t_> src_r_(begin_);
288 for(++src_r_; src_r_ != end_; ++src_r_) {
289 if(!equal_(*dst_r_, *src_r_)) {
290 ++dst_r_;
291 *dst_r_ = hxmove(*src_r_);
292 }
293 }
294 return ++dst_r_;
295}
