33template<typename T_>
34concept hxforward_iterator_concept_ = requires(T_& x_, const T_& y_) {
35 *x_;
36 { ++x_ } -> hxsame_as<T_&>;
37 { x_++ } -> hxsame_as<T_>;
38 { x_ == y_ } -> hxconvertible_to<bool>;
39 { x_ != y_ } -> hxconvertible_to<bool>;
40};
41template<typename T_>
42concept hxbidirectional_iterator_concept_ = hxforward_iterator_concept_<T_> && requires(T_& x_) {
43 { --x_ } -> hxsame_as<T_&>;
44 { x_-- } -> hxsame_as<T_>;
45};
46template<typename T_>
47concept hxrandom_iterator_concept_ = hxbidirectional_iterator_concept_<T_> && requires(
48 const T_& x_, const T_& y_, ptrdiff_t n_) {
49 { x_ + n_ } -> hxsame_as<T_>;
50 { x_ - n_ } -> hxsame_as<T_>;
51 { x_ - y_ } -> hxsame_as<ptrdiff_t>;
52 { x_ < y_ } -> hxconvertible_to<bool>;
53};
54template<typename T_>
55concept hxsorted_iterator_concept_ = hxrandom_iterator_concept_<T_> && requires(const T_& x_) {
56 { x_ < x_ } -> hxconvertible_to<bool>;
57 { x_ == x_ } -> hxconvertible_to<bool>;
58};

231template<hxrange_concept_ range0_t_, hxrange_concept_ range1_t_, typename equal_t_>
232hxattr_nodiscard hxinline hxconstexpr hxattr_flatten
233bool hxequal_range(range0_t_&& range0_, range1_t_&& range1_, const equal_t_& equal_) {
234 hxrestrict_t<decltype(range0_.begin())> begin0_(range0_.begin());
235 const auto end0_ = range0_.end();
236 hxrestrict_t<decltype(range1_.begin())> begin1_(range1_.begin());
237 const auto end1_ = range1_.end();
238#if HX_CPLUSPLUS >= 202002L
239 if constexpr(requires(const decltype(begin0_)& a_, const decltype(begin0_)& b_) { { b_ - a_ }; }
240 && requires(const decltype(begin1_)& a_, const decltype(begin1_)& b_) { { b_ - a_ }; }) {
241 if((end0_ - begin0_) != (end1_ - begin1_)) { return false; }
242 while(begin0_ != end0_) {
243 if(!equal_(*begin0_, *begin1_)) { return false; }
244 ++begin0_; ++begin1_;
245 }
246 return true;
247 }
248 else
249#endif
250 {
251 while(begin0_ != end0_ && begin1_ != end1_) {
252 if(!equal_(*begin0_, *begin1_)) { return false; }
253 ++begin0_; ++begin1_;
254 }
255 return begin0_ == end0_ && begin1_ == end1_;
256 }
257}

347template<hxrandom_range_concept_ range_t_, typename value_t_, typename less_t_>
348hxattr_nodiscard hxinline hxconstexpr hxattr_flatten
349auto hxlower_bound(range_t_&& range_, const value_t_& value_, const less_t_& less_)
350 -> decltype(range_.begin()) {
351 using iterator_t_ = hxrestrict_t<decltype(range_.begin())>;
352 iterator_t_ begin_ = range_.begin();
353 // Does not dereference null pointer args.
354 hxsize_t count_ = range_.end() - begin_;
355 while(count_ > hxsize_t{0}) {
356 const hxsize_t step_ = count_ >> 1;
357 const iterator_t_ mid_ = begin_ + step_;
358 if(less_(*mid_, value_)) {
359 begin_ = mid_ + hxsize_t{1};
360 count_ -= step_ + hxsize_t{1};
361 }
362 else {
363 count_ = step_;
364 }
365 }
366 return begin_;
367}
