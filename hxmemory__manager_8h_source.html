243template <typename T_, hxsystem_allocator_t allocator_=hxsystem_allocator_current,
244 hxalignment_t align_=hxalignment, typename... Args_>
245T_* hxnew(Args_&&... args_) noexcept {
246 // Implements hxforward.
247 return ::new(hxmalloc_ext(sizeof(T_), allocator_, align_)) T_(static_cast<Args_&&>(args_)...);
248}

253template <typename T_>
254void hxdelete(T_* t_) noexcept {
255 if(t_) {
256 t_->~T_();
257 hxfree(const_cast<void*>(static_cast<const void*>(t_)));
258 }
